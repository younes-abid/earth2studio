## Performance Benchmarking

### 1. Memory Usage Analysis

```bash
# Monitor GPU memory during a test run
python -c "
import torch
import time

if torch.cuda.is_available():
    print('🔍 GPU Memory Analysis:')
    
    # Clear cache
    torch.cuda.empty_cache()
    
    for i in range(torch.cuda.device_count()):
        torch.cuda.set_device(i)
        
        # Before
        mem_before = torch.cuda.memory_allocated() / 1e9
        mem_cached_before = torch.cuda.memory_reserved() / 1e9
        
        print(f'GPU {i} - Before: Allocated {mem_before:.2f}GB, Cached {mem_cached_before:.2f}GB')
        
        # Simulate model loading (without actually running inference)
        try:
            # This will show peak memory usage during model initialization
            from earth2studio.models.dx import CorrDiffTaiwan
            print('  Loading CorrDiff model...')
            
            package = CorrDiffTaiwan.load_default_package()
            # Note: This downloads ~2GB model if not cached
            
            mem_after = torch.cuda.memory_allocated() / 1e9
            mem_cached_after = torch.cuda.memory_reserved() / 1e9
            
            print(f'GPU {i} - After: Allocated {mem_after:.2f}GB, Cached {mem_cached_after:.2f}GB')
            print(f'  Model Memory Usage: {mem_after - mem_before:.2f}GB')
            
        except Exception as e:
```

## Troubleshooting Common Issues

### Model Download Failures

```bash
# If NGC download fails
python -c "
import os
from earth2studio.models.auto import Package

# Check cache location
cache_dir = Package.default_cache('corrdiff_taiwan')
print(f'Model cache directory: {cache_dir}')
print(f'Cache exists: {os.path.exists(cache_dir)}')

if os.path.exists(cache_dir):
    print('Cache contents:')
    for item in os.listdir(cache_dir):
        path = os.path.join(cache_dir, item)
        size = os.path.getsize(path) / 1e6 if os.path.isfile(path) else 'DIR'
        print(f'  {item}: {size}MB' if isinstance(size, float) else f'  {item}: {size}')
"
```

### Memory Issues

```bash
# If running out of GPU memory
python -c "
import torch

if torch.cuda.is_available():
    props = torch.cuda.get_device_properties(0)
    total_memory = props.total_memory / 1e9
    
    print(f'Total GPU Memory: {total_memory:.1f}GB')
    
    if total_memory < 8:
        print('⚠️  Warning: CorrDiff may require >8GB GPU memory')
        print('   Consider reducing number_of_samples or using CPU fallback')
    else:
        print('✅ GPU memory should be sufficient')
```
